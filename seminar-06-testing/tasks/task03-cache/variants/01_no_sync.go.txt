// В этой реализации нет никакой синхронизации: конкурентные Set рвут
// мапу (паника «concurrent map writes» или потерянные обновления).
// Тесты должны её отклонить — стресс-тестом из многих горутин.

package cache

type Cache struct {
	items map[string]string
}

func New() *Cache {
	return &Cache{items: make(map[string]string)}
}

func (c *Cache) Get(key string) (string, bool) {
	v, ok := c.items[key]
	return v, ok
}

func (c *Cache) Set(key, value string) {
	c.items[key] = value
}

func (c *Cache) Len() int {
	return len(c.items)
}
