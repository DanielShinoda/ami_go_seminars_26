// В этой реализации все кеши делят одно хранилище: запись, сделанная
// через один экземпляр Cache, видна из любого другого. Тесты должны её
// отклонить.

package cache

import "sync"

var (
	globalMu    sync.RWMutex
	globalItems = make(map[string]string) // ← общая мапа на все экземпляры
)

type Cache struct{}

func New() *Cache {
	return &Cache{}
}

func (c *Cache) Get(key string) (string, bool) {
	globalMu.RLock()
	defer globalMu.RUnlock()
	v, ok := globalItems[key]
	return v, ok
}

func (c *Cache) Set(key, value string) {
	globalMu.Lock()
	defer globalMu.Unlock()
	globalItems[key] = value
}

func (c *Cache) Len() int {
	globalMu.RLock()
	defer globalMu.RUnlock()
	return len(globalItems)
}
